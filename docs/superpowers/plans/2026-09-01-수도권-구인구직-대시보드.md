> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 서울고용센터에 보고서와 함께 넘길 모바일 대시보드를, 매달 스스로 갱신되고 틀린 값이면 배포되지 않는 파이프라인 위에 세운다.

**Architecture:** 파이썬 수집기가 KOSIS OpenAPI와 EIS OLAP에서 받아 집계 JSON을 만들고, 검사를 통과할 때만 커밋한다. 화면은 빌드 없는 정적 ES 모듈로 그 JSON만 읽는다. 고용데이터아카이브의 토큰·앱 셸을 그대로 쓴다.

**Tech Stack:** Python 3.11 · requests · playwright · pytest / 바닐라 ES 모듈 · IBM Plex Sans KR / GitHub Actions · GitHub Pages

**Spec:** `docs/superpowers/specs/2026-09-01-수도권-구인구직-대시보드-design.md`

## Global Constraints

- 저장소 루트는 `사무직감소\06_대시보드\`. 상위 폴더의 보고서·원자료는 저장소에 넣지 않는다.
- 테스트는 **네트워크에 나가지 않는다.** 모든 수집기는 `fetch` 함수를 주입받고, 테스트는 저장된 샘플을 준다.
- KOSIS 키는 환경변수 `KOSIS_API_KEY`. 저장소에 넣지 않는다.
- KOSIS 반기 주기 코드는 `prdSe="S"` (`H` 아님).
- 시군구 **70개**, 센터 **39개**. 이중배정 0.
- **유효구직건수의 분해값을 더해 총계로 쓰지 않는다.** 총계는 총계 행에서 받는다.
- **유효는 월별 합산 금지.**
- 경기 일반구 코드(`41111` 등) 사용 금지. 폐지 코드 `41283`·`41710`·`41730`·`41810` 제외.
- 색: 구인 `#1baf7a` · 구직 `#2a78d6` · 직종별사업체노동력조사 `#eb6834` · accent `#23508f`.
  지도 램프 — 서울 `#e6f6f0 #b6e6d3 #7ed4b3 #45c194 #1baf7a`, 경기 `#e6effa #c2d8f4 #8fb6ea #5a97e0 #2a78d6`, 인천 `#f6e9f2 #e6c6dd #d29ec3 #c473a9 #b5478f`.
- 라이트 전용(`color-scheme: light`). 앱 셸 `max-width: 480px`.
- 카드 제목은 선택 대상 이름으로 시작하고 **12자를 넘으면 줄인다**.
- **제공되지 않는 값은 카드째 감춘다.** 빈칸도, 전국 값 대체도 하지 않는다.

# pipeline/center_map.py
"""시군구를 센터로 묶는다.

이 매핑은 화면과 수집기가 함께 쓰는 유일한 지역 계약이다. 시군구 하나가 센터 둘에
붙으면 센터별 합계가 이중계상되므로 load 시점에 반드시 validate 한다.
"""
from __future__ import annotations

import json
from dataclasses import dataclass
from pathlib import Path


@dataclass(frozen=True)
class CenterMap:
    _by_code: dict[str, str]
    _centers: tuple[str, ...]

    def center_of(self, sigungu_code: str) -> str:
        return self._by_code[sigungu_code]

    def codes(self) -> set[str]:
        return set(self._by_code)

    def centers(self) -> list[str]:
        return list(self._centers)

    def validate(self) -> None:
        if len(self._by_code) != 70:
            raise ValueError(f"시군구가 70개가 아니다: {len(self._by_code)}")
        if len(self._centers) != len(set(self._centers)):
            raise ValueError("센터 이름이 중복된다")


def load(path: Path) -> CenterMap:
    raw = json.loads(Path(path).read_text(encoding="utf-8"))
    by_code: dict[str, str] = {}
    centers: list[str] = []
    for entry in raw["센터"]:
        name = entry["센터"]
        centers.append(name)
        for gu in entry["시군구"]:
            code = gu["code"]
            if code in by_code:
                raise ValueError(f"이중배정: {code} → {by_code[code]}, {name}")
            by_code[code] = name
    cm = CenterMap(by_code, tuple(centers))
    cm.validate()
    return cm
```

# pipeline/kosis.py
"""KOSIS OpenAPI 클라이언트.

주의: 직종별사업체노동력조사는 반기 자료이고 주기 코드는 'S' 다. 'H' 로 부르면
KOSIS 가 '필수요청변수값이 누락되었습니다'(err 20) 를 돌려준다 — 원인을 짚기
어려운 오류라 여기 한 곳에 못 박아 둔다.
"""
from __future__ import annotations

import requests

API = "https://kosis.kr/openapi/statisticsData.do"
ORG_ID = "118"


class KosisError(RuntimeError):
    pass


def fetch(table, *, item, obj_l1, obj_l2, obj_l3,
          periods=None, recent=None, api_key, get=requests.get):
    params = {
        "method": "getList", "apiKey": api_key, "orgId": ORG_ID, "tblId": table,
        "itmId": item, "objL1": obj_l1, "objL2": obj_l2, "objL3": obj_l3,
        "prdSe": "S", "format": "json", "jsonVD": "Y",
    }
    if periods:
        params["startPrdDe"], params["endPrdDe"] = periods
    else:
        params["newEstPrdCnt"] = str(recent or 1)

    payload = get(API, params=params, timeout=120).json()
    if isinstance(payload, dict):
        raise KosisError(payload.get("errMsg", str(payload)))
    return payload
```

"""EIS 리포트의 현재 뷰어 주소를 찾는다.

reportId 는 EIS 가 재발급할 수 있어 하드코딩하지 않는다. 대신 메뉴 페이지가
숨은 input 'reptIdUrl' 에 현재 주소를 들고 있으므로 그것을 읽는다.

from __future__ import annotations

import html as html_mod
import json
import re
from pathlib import Path

import requests

BASE = "https://eis.work24.go.kr/eisps/rpt/reptDtl.do"
UA = {"User-Agent": "Mozilla/5.0 (compatible; seoul-jobs-dashboard/1.0)"}

_CONFIG = json.loads((Path(__file__).resolve().parents[1] / "config/olap.json")
                     .read_text(encoding="utf-8"))
REPORTS: dict[str, str] = _CONFIG["reports"]

_PATTERN = re.compile(r"id=\"reptIdUrl\"[^>]*value='([^']+)'")


class EisReportError(RuntimeError):
    pass


def viewer_url(menu_id: str, *, get=requests.get) -> str:
    page = get(f"{BASE}?menuId={menu_id}", headers=UA, timeout=60).text
    found = _PATTERN.search(page)
    if not found:
        raise EisReportError(f"menuId={menu_id} 페이지에 reptIdUrl 이 없다 — 화면 개편 의심")
    return html_mod.unescape(found.group(1))

# pipeline/olap.py (DOM 인 경우)
"""렌더된 OLAP 그리드를 DOM 에서 읽는다.

cube 응답이 세션 AES 로 암호화돼 있어 평문 파싱이 불가능할 때 쓰는 길이다.
가상화 때문에 보이는 행만 오므로, 스크롤을 끝까지 내리며 모은다.
"""
from __future__ import annotations


def parse_grid(rows_text: list[list[str]]) -> list[dict]:
    header, *body = rows_text
    return [dict(zip(header, row)) for row in body]


def fetch_grid(url: str, *, page, max_scrolls: int = 200) -> list[list[str]]:
    page.goto(url, wait_until="networkidle", timeout=90_000)
    page.wait_for_selector("div.dx-datagrid-rowsview", timeout=60_000)

    seen: dict[str, list[str]] = {}
    for _ in range(max_scrolls):
        rows = page.eval_on_selector_all(
            "div.dx-datagrid-rowsview tr",
            "els => els.map(tr => Array.from(tr.cells).map(td => td.innerText.trim()))",
        )
        before = len(seen)
        for row in rows:
            if row:
                seen["".join(row)] = row
        page.mouse.wheel(0, 2000)
        page.wait_for_timeout(250)
        if len(seen) == before:
            break
    return list(seen.values())
```

class UnknownRegion(ValueError):
    pass


def _period(text: str) -> str:
    found = re.search(r"(\d{4})년\s*(\d{2})월", text)
    if not found:
        raise ValueError(f"기간을 못 읽는다: {text}")
    return found.group(1) + found.group(2)


def _number(text):
    text = (text or "").strip().replace(",", "")
    return 0 if text in ("", "-") else int(float(text))


def _code(name: str) -> str:
    try:
        return SIGUNGU_NAME_TO_CODE[name.strip()]
    except KeyError:
        raise UnknownRegion(f"매핑에 없는 지역: {name}") from None


def _strip_prefix(text: str) -> str:
    return re.sub(r"^\d{4}직종_", "", (text or "").strip())


def collect_vacancy(rows, cm):
    out = []
    for row in rows:
        code = _code(row["(근무지역)시군구"])
        out.append({
            "period": _period(row["마감년월"]),
            "sigungu": code,
            "center": cm.center_of(code),
            "occupation": _strip_prefix(row.get("직종_중분류", "")),
            "industry": (row.get("산업_대분류") or "").strip(),
            "vacancy": _number(row.get("유효구인인원(전체)")),
            "seekers": _number(row.get("유효구직건수(전체)")),
        })
    return out


def collect_placement(rows, cm):
    out = []
    for row in rows:
        code = _code(row["(근무지역)시군구"])
        out.append({

            "sigungu": code,
            "center": cm.center_of(code),

            "placements": _number(row.get("취업건수(월)")),

def collect_insured(rows, cm):
    out = []
    for row in rows:
        code = _code(row["(사업장)시군구"])
        out.append({
            "period": _period(row["마감년월"]),

            "insured": _number(row.get("피보험자수(전체)")),
            "gained": _number(row.get("취득자수(월)")),
            "lost": _number(row.get("상실자수(월)")),

def collect_mobility(rows):
    return [{
        "period": _period(row["마감년월"]),
        "sido": (row.get("(사업장)시도") or "").strip(),

        "prev_industry": (row.get("산업(이전)_대분류") or "").strip(),
        "movers": _number(row.get("경력이동자수(월)")),
    } for row in rows]

# pipeline/checks.py
"""수집 결과가 배포돼도 되는지 본다.

이 파일의 목적은 하나다 — 조용히 틀린 값이 화면에 오르는 것을 막는다.
어긋나면 예외를 던지고, 호출부는 커밋하지 않는다.
"""
from __future__ import annotations

INCHEON_OLD_TO_NEW = {
    "28110": ["28125", "28155"],   # 중구 → 제물포·영종
    "28140": ["28125"],            # 동구 → 제물포
    "28260": ["28275", "28290"],   # 서구 → 서해·검단
}

SEAM_RATIO = 2.0  # 표 경계에서 이보다 크게 튀면 코드 매핑 의심


class CheckFailed(RuntimeError):
    pass


def check_regions(rows, cm) -> None:
    seen = {row["sigungu"] for row in rows}
    missing = cm.codes() - seen
    if missing:
        raise CheckFailed(f"시군구 {len(missing)}개가 비었다: {sorted(missing)[:5]} …")


def check_not_all_zero(rows, field) -> None:
    if not any(row.get(field) for row in rows):
        raise CheckFailed(f"{field} 가 전부 0 이다 — 수집이 비었다")


def check_not_identical_to_previous(rows, previous) -> None:
    if previous and rows == previous:
        raise CheckFailed("전월과 완전히 동일하다 — 새 자료가 아니다")


def check_incheon_codes(rows) -> None:
    by_period: dict[str, set[str]] = {}
    for row in rows:
        by_period.setdefault(row["period"], set()).add(row["sigungu"])
    for period, codes in by_period.items():
        for old, news in INCHEON_OLD_TO_NEW.items():
            overlap = [n for n in news if n in codes]
            if old in codes and overlap:
                raise CheckFailed(
                    f"{period}: 개편 전 {old} 와 신설 {overlap} 가 함께 있다 — 더하면 이중계상")


def check_est_seam(rows) -> None:
    by_key: dict[tuple, dict[str, int]] = {}
    for row in rows:
        by_key.setdefault((row["occupation"], row["item"]), {})[row["period"]] = row["value"]
    for key, series in by_key.items():
        if "202502" in series and "202601" in series:
            before, after = series["202502"], series["202601"]
            if before and (after / before > SEAM_RATIO or before / max(after, 1) > SEAM_RATIO):
                raise CheckFailed(f"표 경계에서 값이 튄다 {key}: {before} → {after}")
```

"""수집을 엮는다.

원칙: 검사를 다 통과한 뒤에야 파일을 쓴다. 절반만 갱신된 상태가 가장 나쁘다.

from __future__ import annotations

import json
from datetime import datetime, timezone
from pathlib import Path

from pipeline import checks

FIELD_OF = {"vacancy": "vacancy", "placement": "placements",
            "insured": "insured", "mobility": "movers"}


def run_monthly(period, *, out_dir, fetchers, cm, previous=None):
    collected = {name: fetch(period) for name, fetch in fetchers.items()}

    for name, rows in collected.items():
        if name in ("vacancy", "placement", "insured"):
            checks.check_regions(rows, cm)
            checks.check_incheon_codes(rows)
        checks.check_not_all_zero(rows, FIELD_OF[name])
        checks.check_not_identical_to_previous(rows, (previous or {}).get(name))

    out_dir = Path(out_dir)
    out_dir.mkdir(parents=True, exist_ok=True)
    stamp = datetime.now(timezone.utc).isoformat(timespec="seconds")
    for name, rows in collected.items():
        (out_dir / f"{name}.json").write_text(
            json.dumps({"period": period, "collected_at": stamp, "rows": rows},
                       ensure_ascii=False),
            encoding="utf-8")
    return {name: len(rows) for name, rows in collected.items()}


def run_halfyear(period, *, out_dir, api_key, collector=None):
    from pipeline import est

    collector = collector or est.collect
    rows = collector([period], api_key=api_key)
    checks.check_est_seam(rows)
    checks.check_not_all_zero(rows, "value")

    out_dir = Path(out_dir)
    out_dir.mkdir(parents=True, exist_ok=True)
    (out_dir / "est.json").write_text(
        json.dumps({"period": period, "rows": rows}, ensure_ascii=False),
        encoding="utf-8")
    return {"est": len(rows)}

// app/js/data.js
const MAX_TITLE = 12;

export function titleFor(name, suffix) {
  const head = name.length > MAX_TITLE ? name.slice(0, MAX_TITLE) + "…" : name;
  return suffix ? `${head} ${suffix}` : head;
}

export function ratio(vacancy, seekers) {
  if (!seekers) return null;
  return Math.round((vacancy / seekers) * 100) / 100;
}

const FILES = ["vacancy", "placement", "insured", "mobility", "est", "center_map"];

export async function load(base = "../data") {
  const entries = await Promise.all(FILES.map(async (name) => {
    const res = await fetch(`${base}/${name}.json`);
    if (!res.ok) throw new Error(`${name}.json 을 못 읽었다 (${res.status})`);
    return [name, await res.json()];
  }));
  return Object.fromEntries(entries);
}

function bars(rows, { label, value, highlight }) {
  const max = Math.max(...rows.map((r) => r[value]), 1);
  return `<div class="bars">${rows.map((r) => `
    <div class="bar${r[label] === highlight ? " bar--hi" : ""}">
      <span class="bar__lab">${r[label]}</span>
      <span class="bar__val num">${num(r[value])}</span>
      <i class="bar__track" style="width:${Math.round((r[value] / max) * 100)}%"></i>
    </div>`).join("")}</div>`;
}

export function render(store, selection) {
  const { sido, industry } = selection;
  const ranking = store.vacancy.rows
    .filter((r) => r.sido === sido && r.industry)
    .sort((a, b) => b.vacancy - a.vacancy)
    .slice(0, 8);
  const ins = store.insured.rows.find((r) => r.sido === sido && r.industry === industry) ?? {};
  const moves = store.mobility.rows
    .filter((r) => r.industry === industry)
    .sort((a, b) => b.movers - a.movers)
    .slice(0, 5);
  const hire = store.est.rows.find((r) =>
    r.sido === sido && r.size === "전규모" && r.industry === industry && r.item === "채용인원");

  // 산업 탭에 구직은 없다 — 유효구직에 산업 축 자체가 없다(수집하지 않는 항목).
  const cards = [

           body: bars(ranking, { label: "industry", value: "vacancy", highlight: industry }) }),

           body: `<div class="card__value num">${num(ins.insured)}<small>명</small></div>` }),
  ];

  if (moves.length) {
    cards.push(card({
      title: titleFor(industry, "경력직 이동"),

      body: `<div class="card__note" style="margin:0">유입 경력직의 <b>이전 업종</b></div>

// app/js/tilemap.js
const RAMP = { "11": "s", "31": "g", "28": "i" };

export function step(value, breaks) {
  for (let i = 0; i < breaks.length; i += 1) {
    if (value < breaks[i]) return i + 1;
  }
  return breaks.length + 1;
}

export function render(values, layout, { breaks = [0.08, 0.12, 0.18, 0.25], selected } = {}) {
  const tiles = Object.entries(layout).map(([code, cell]) => {
    const value = values[code];
    const ramp = RAMP[cell.sido];
    const level = value == null ? 1 : step(value, breaks);
    const on = code === selected ? " on" : "";
    return `<i class="tile${on}" title="${cell.name ?? code}" data-code="${code}"
      style="grid-area:${cell.row}/${cell.col}; background:var(--${ramp}${level})"></i>`;
  });
  return `<div class="tilemap">${tiles.join("")}</div>`;
}

const period = (p) => p.replace(/(\d{4})(\d{2})/, "$1.$2");

function butterfly(rows) {
  // 좌우 같은 눈금. 눈금을 나누면 "구인이 구직만큼 있다"는 거짓 인상이 생긴다.

  return `<div class="bf">
    <div class="bf__head l">유효구인(명)</div><div class="bf__head">센터</div><div class="bf__head r">유효구직(건)</div>

      <div class="bf__c${r.selected ? " is-sel" : ""}">${r.center.replace("고용센터", "")}</div>
      <div class="bf__r"><i style="width:${width(r.seekers)}"></i><span class="bf__n num">${num(r.seekers)}</span></div>
    `).join("")}
  </div>`;

export function render(store, selection, layout = {}) {
  const rows = store.vacancy.rows;
  const values = Object.fromEntries(rows.map((r) => [r.sigungu, ratio(r.vacancy, r.seekers)]));
  const picked = rows.find((r) => r.sigungu === selection.sigungu);

  const byCenter = new Map();
  rows.forEach((r) => {
    const acc = byCenter.get(r.center) ?? { center: r.center, vacancy: 0, seekers: 0 };
    acc.vacancy += r.vacancy;
    acc.seekers += r.seekers;
    byCenter.set(r.center, acc);
  });
  const centers = [...byCenter.values()]
    .sort((a, b) => b.vacancy - a.vacancy)
    .slice(0, 8)
    .map((c) => ({ ...c, selected: c.center === picked?.center }));

  const panel = picked ? `<div class="mapsel">
      <div class="mapsel__head"><span class="gu">${picked.sigungu}</span> › ${picked.center}</div>
      <dl class="mapsel__stats">
        <div><dt>유효구인</dt><dd class="num">${num(picked.vacancy)}<small>명</small></dd></div>
        <div><dt>유효구직</dt><dd class="num">${num(picked.seekers)}<small>건</small></dd></div>
        <div><dt>구인배수</dt><dd class="num">${ratio(picked.vacancy, picked.seekers)}</dd></div>
      </dl>
    </div>` : "";

  return `<div class="cards">
    ${card({ title: "구인배수 지도", badge: period(store.vacancy.period),
             body: renderMap(values, layout, { selected: selection.sigungu }) + panel,

             body: butterfly(centers) })}
  </div>`;
}

```python
# pipeline/cli.py
"""워크플로 진입점. 검사가 실패하면 0 이 아닌 코드로 죽어 커밋 단계가 실행되지 않는다."""
import os
import sys
from datetime import date
from pathlib import Path

from pipeline import center_map, collect

ROOT = Path(__file__).resolve().parents[1]


def latest_month() -> str:
    today = date.today()
    year, month = (today.year, today.month - 2) if today.month > 2 else (today.year - 1, today.month + 10)
    return f"{year}{month:02d}"


def main(mode: str) -> int:
    cm = center_map.load(ROOT / "data/center_map.json")
    if mode == "monthly":
        from pipeline import fetchers  # Task 7 의 수집기를 묶은 얇은 모듈
        summary = collect.run_monthly(latest_month(), out_dir=ROOT / "data",
                                      fetchers=fetchers.MONTHLY, cm=cm)
    elif mode == "halfyear":
        period = f"{date.today().year}{'01' if date.today().month < 7 else '02'}"
        summary = collect.run_halfyear(period, out_dir=ROOT / "data",
                                       api_key=os.environ["KOSIS_API_KEY"])
    else:
        raise SystemExit(f"모르는 모드: {mode}")
    print(summary)
    return 0


if __name__ == "__main__":
    sys.exit(main(sys.argv[1]))
```
